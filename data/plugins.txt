# Плагины стика.
#
# Файл читается при старте. Чтобы применить правки, пересобирать прошивку не
# нужно — достаточно залить файловую систему:
#
#     pio run -e esp32dev -t uploadfs
#
# Формат: блоки, разделённые пустой строкой; строки "ключ = значение";
# всё после # — комментарий.
#
#   name        имя функции для модели (латиницей, без пробелов)
#   description что делает — по этому модель решает, когда вызывать
#   kind        gpio | screen | http
#   pin         для gpio: номер пина
#   url         для http: адрес (https)
#   answer      true — попросить модель ещё и прокомментировать результат
#
# Схема аргументов задаётся видом действия: gpio ждёт state (on/off/toggle),
# screen — text, http — необязательный query.

name = set_lamp
description = Включить, выключить или переключить лампу на GPIO26. Вызывай всегда, когда просят включить, выключить или переключить лампу.
kind = gpio
pin = 26

name = show_text
description = Показать короткий текст на экране стика. Вызывай, когда просят показать что-то на экране.
kind = screen

name = http_get
description = Отправить HTTP GET на домашний адрес (умный дом, вебхук). query — необязательный параметр строки запроса.
kind = http
url = https://example.com/hook

name = weather
description = Узнать погоду на сегодня в городе. Вызывай, когда спрашивают про погоду. Если город не назван — вызови без параметра city.
kind = weather

name = switch
description = Включить, выключить или переключить выключатель. channel: 1, 2, 3 — конкретный канал, all — весь свет сразу. Вызывай всегда, когда просят включить, выключить или переключить свет или выключатель.
kind = tuya
